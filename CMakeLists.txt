cmake_minimum_required(VERSION 3.20)
project(enevina_lite LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(enevina_core physics/physics.cpp physics/calibration.cpp optimizer/optimizer.cpp forecast/pricing.cpp forecast/validity.cpp forecast/sources.cpp forecast/fmi.cpp safety/supervisor.cpp pump/heishamon_adapter.cpp)
target_include_directories(enevina_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(enevina_core PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE enevina_core)
add_test(NAME core_tests COMMAND core_tests)
add_executable(enevina_cli tools/enevina_cli.cpp)
target_link_libraries(enevina_cli PRIVATE enevina_core)
if(WIN32)
  target_link_libraries(enevina_cli PRIVATE psapi)
endif()

# Platform-independent P0 regression tests, including ESP weather-cache format.
add_executable(optimizer_regression_tests tests/optimizer_regression_tests.cpp)
target_link_libraries(optimizer_regression_tests PRIVATE enevina_core)
add_test(NAME optimizer_regression_tests COMMAND optimizer_regression_tests)
add_executable(weather_cache_tests tests/weather_cache_tests.cpp)
target_include_directories(weather_cache_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME weather_cache_tests COMMAND weather_cache_tests)
