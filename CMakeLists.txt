cmake_minimum_required(VERSION 3.20)
project(enevina_lite LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(enevina_core physics/physics.cpp physics/calibration.cpp optimizer/optimizer.cpp forecast/pricing.cpp forecast/validity.cpp safety/supervisor.cpp pump/heishamon_adapter.cpp)
target_include_directories(enevina_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(enevina_core PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE enevina_core)
add_test(NAME core_tests COMMAND core_tests)
add_executable(enevina_cli tools/enevina_cli.cpp)
target_link_libraries(enevina_cli PRIVATE enevina_core)
if(WIN32)
  target_link_libraries(enevina_cli PRIVATE psapi)
endif()
